cmake_minimum_required(VERSION 3.16)
project(cortexlite VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(cortexlite_c STATIC src/c/tensor.c src/c/inference.c)
target_include_directories(cortexlite_c PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_compile_options(cortexlite_c PRIVATE -Wall -Wextra -Wpedantic)

add_library(cortexlite_dsa STATIC src/cpp/vector_index.cpp)
target_include_directories(cortexlite_dsa PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(cortexlite_dsa PUBLIC cortexlite_c)
target_compile_options(cortexlite_dsa PRIVATE -Wall -Wextra -Wpedantic)

add_executable(cortexlite src/cpp/main.cpp)
target_link_libraries(cortexlite PRIVATE cortexlite_dsa)
target_compile_options(cortexlite PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
  add_executable(cortexlite_tests tests/test_core.cpp)
  target_link_libraries(cortexlite_tests PRIVATE cortexlite_dsa)
  add_test(NAME unit COMMAND cortexlite_tests "${PROJECT_SOURCE_DIR}/data/demo_model.clm")
  add_test(NAME evaluate COMMAND cortexlite evaluate --model "${PROJECT_SOURCE_DIR}/data/demo_model.clm" --samples "${PROJECT_SOURCE_DIR}/data/samples.csv" --eval "${PROJECT_SOURCE_DIR}/data/eval.csv")
  add_test(NAME graph_evaluate COMMAND cortexlite evaluate --model "${PROJECT_SOURCE_DIR}/data/demo_model.clm" --samples "${PROJECT_SOURCE_DIR}/data/samples.csv" --eval "${PROJECT_SOURCE_DIR}/data/eval.csv" --search graph --ef 32)
endif()
