cmake_minimum_required(VERSION 3.16)
project(cortexlite VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CORTEXLITE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(CORTEXLITE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizers require GCC or Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_library(cortexlite_c STATIC src/c/tensor.c src/c/inference.c)
target_include_directories(cortexlite_c PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_compile_options(cortexlite_c PRIVATE -Wall -Wextra -Wpedantic)

add_library(cortexlite_dsa STATIC src/cpp/vector_index.cpp)
target_include_directories(cortexlite_dsa PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(cortexlite_dsa PUBLIC cortexlite_c)
target_compile_options(cortexlite_dsa PRIVATE -Wall -Wextra -Wpedantic)

find_package(Threads REQUIRED)
add_executable(cortexlite src/cpp/main.cpp)
target_link_libraries(cortexlite PRIVATE cortexlite_dsa Threads::Threads)
target_compile_options(cortexlite PRIVATE -Wall -Wextra -Wpedantic)

add_executable(cortexlite_vector_benchmark benchmarks/vector_benchmark.cpp)
target_link_libraries(cortexlite_vector_benchmark PRIVATE cortexlite_dsa)
include(CTest)
if(BUILD_TESTING)
  add_executable(cortexlite_tests tests/test_core.cpp)
  target_link_libraries(cortexlite_tests PRIVATE cortexlite_dsa)
  add_test(NAME unit COMMAND cortexlite_tests "${PROJECT_SOURCE_DIR}/data/demo_model.clm")
  add_test(NAME evaluate COMMAND cortexlite evaluate --model "${PROJECT_SOURCE_DIR}/data/demo_model.clm" --samples "${PROJECT_SOURCE_DIR}/data/samples.csv" --eval "${PROJECT_SOURCE_DIR}/data/eval.csv")
  add_test(NAME int8_evaluate COMMAND cortexlite evaluate --model "${PROJECT_SOURCE_DIR}/data/demo_model.clm" --samples "${PROJECT_SOURCE_DIR}/data/samples.csv" --eval "${PROJECT_SOURCE_DIR}/data/eval.csv" --precision int8)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|aarch64|arm64)$")
    add_test(NAME simd_evaluate COMMAND cortexlite evaluate --model "${PROJECT_SOURCE_DIR}/data/demo_model.clm" --samples "${PROJECT_SOURCE_DIR}/data/samples.csv" --eval "${PROJECT_SOURCE_DIR}/data/eval.csv" --kernel simd)
  endif()
  add_test(NAME multithread_benchmark COMMAND cortexlite benchmark --model "${PROJECT_SOURCE_DIR}/data/demo_model.clm" --samples "${PROJECT_SOURCE_DIR}/data/samples.csv" --eval "${PROJECT_SOURCE_DIR}/data/eval.csv" --threads 4 --iterations 128)
  add_test(NAME vector_benchmark_smoke COMMAND cortexlite_vector_benchmark --vectors 128 --dimensions 16 --queries 16 --k 5 --ef 32)
  add_test(NAME graph_evaluate COMMAND cortexlite evaluate --model "${PROJECT_SOURCE_DIR}/data/demo_model.clm" --samples "${PROJECT_SOURCE_DIR}/data/samples.csv" --eval "${PROJECT_SOURCE_DIR}/data/eval.csv" --search graph --ef 32)
endif()
